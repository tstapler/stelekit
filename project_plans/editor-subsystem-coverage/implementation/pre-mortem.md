## Gate script for failure #1 (run after Task 1.3a)
```
python3 -I - <<'PY'
import re, subprocess
base = {f.rsplit("/", 1)[-1] for f in subprocess.check_output(["git", "ls-files"], text=True).split("\n")}
ids = re.findall(r"<ID>[A-Za-z]+:([A-Za-z0-9_]+\.kt)\$", open("kmp/config/detekt/baseline.xml").read())
stale = sorted({i for i in ids if i not in base})
print("stale baseline files:", stale)
raise SystemExit(1 if stale else 0)
PY
```
